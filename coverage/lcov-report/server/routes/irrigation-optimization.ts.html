  try {

      crop_type = "wheat",
      current_moisture = 50.0,
      field_capacity = 35.0,
      wilting_point = 15.0,
    } = req.body;
 
    const available_water = current_moisture - wilting_point;
    const max_available_water = field_capacity - wilting_point;
    const water_stress_level = Math.max(
